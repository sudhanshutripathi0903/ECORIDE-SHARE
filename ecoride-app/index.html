// ============================================ // CarpoolGo — Session helper // ============================================ export function getCurrentUser() { const raw = localStorage.getItem('carpoolgo_user'); if (!raw) return null; try { return JSON.parse(raw); } catch { return null; } } export function isLoggedIn() { return getCurrentUser() !== null; } export function logout() { localStorage.removeItem('carpoolgo_user'); window.location.href = 'login.html'; } // Guard: redirect to login if not authenticated export function requireAuth() { if (!isLoggedIn()) { window.location.href = 'login.html'; return false; } return true; } // Update any element with [data-user-phone] attribute export function paintUserInfo() { const user = getCurrentUser(); if (!user) return; document.querySelectorAll('[data-user-phone]').forEach(el => { el.textContent = user.phone || ''; }); }
